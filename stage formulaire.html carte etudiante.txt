    <script>
        // Données injectées côté serveur (tag non échappé requis pour du JSON dynamique)
        const ECOLES_DATA = <?!= JSON.stringify(ecolesData) ?>;
        const WEB_APP_URL = <?!= JSON.stringify(webAppUrl) ?>;

        const ecoleSelect = document.getElementById('ecole');
        const formationSelect = document.getElementById('formation');

        function populateEcoles() {
            ecoleSelect.innerHTML = '<option value="">-- Choisir --</option>';
            Object.keys(ECOLES_DATA).forEach(function(nomEcole) {
                const opt = document.createElement('option');
                opt.value = nomEcole;
                opt.textContent = nomEcole;
                ecoleSelect.appendChild(opt);
            });
        }

        function populateFormations() {
            const ecole = ecoleSelect.value;
            formationSelect.innerHTML = '<option value="">-- Choisir --</option>';
            if (!ecole || !ECOLES_DATA[ecole]) return;
            ECOLES_DATA[ecole].formations.forEach(function(f) {
                const opt = document.createElement('option');
                opt.value = f;
                opt.textContent = f;
                formationSelect.appendChild(opt);
            });
        }

        ecoleSelect.addEventListener('change', populateFormations);
        populateEcoles();

        function readFileAsDataUrl(fileInput) {
            return new Promise(function(resolve, reject) {
                const file = fileInput.files[0];
                if (!file) { resolve(null); return; }
                const reader = new FileReader();
                reader.onload = function() { resolve({ data: reader.result }); };
                reader.onerror = reject;
                reader.readAsDataURL(file);
            });
        }

        async function submitForm() {
            const statusMsg = document.getElementById('statusMsg');
            const submitBtn = document.getElementById('submitBtn');

            const requiredFields = ['ecole', 'formation', 'nom', 'prenom', 'matricule', 'email', 'dateNaissance', 'annee'];
            for (const id of requiredFields) {
                if (!document.getElementById(id).value) {
                    statusMsg.textContent = "Veuillez remplir tous les champs obligatoires.";
                    statusMsg.className = "text-sm text-center mt-2 text-red-600 font-semibold";
                    return;
                }
            }

            const photoFile = document.getElementById('photo').files[0];
            const preuveFile = document.getElementById('preuve').files[0];
            const justificatifFile = document.getElementById('justificatif').files[0];

            if (!photoFile || !preuveFile || !justificatifFile) {
                statusMsg.textContent = "Veuillez joindre la photo, la preuve d'inscription et le justificatif de paiement.";
                statusMsg.className = "text-sm text-center mt-2 text-red-600 font-semibold";
                return;
            }

            submitBtn.disabled = true;
            submitBtn.textContent = "Envoi en cours...";

            try {
                const [photo, preuve, justificatif] = await Promise.all([
                    readFileAsDataUrl(document.getElementById('photo')),
                    readFileAsDataUrl(document.getElementById('preuve')),
                    readFileAsDataUrl(document.getElementById('justificatif'))
                ]);

                const payload = {
                    ecole: ecoleSelect.value,
                    formation: formationSelect.value,
                    nom: document.getElementById('nom').value,
                    prenom: document.getElementById('prenom').value,
                    matricule: document.getElementById('matricule').value,
                    email: document.getElementById('email').value,
                    telephone: document.getElementById('telephone').value,
                    dateNaissance: document.getElementById('dateNaissance').value,
                    annee: document.getElementById('annee').value,
                    photo: photo,
                    preuve: preuve,
                    justificatif: justificatif
                };

                // Content-Type text/plain volontaire : évite le preflight CORS sur les webapps Apps Script
                const response = await fetch(WEB_APP_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                    body: JSON.stringify(payload)
                });

                const result = await response.json();

                if (result.status === 'success') {
                    document.getElementById('formCard').innerHTML =
                        '<div class="text-center py-6"><h2 class="text-xl font-bold text-green-600 mb-2">Demande envoyée !</h2><p class="text-gray-600">Votre dossier est en cours de vérification. Vous recevrez votre carte par email une fois validé.</p></div>';
                } else {
                    statusMsg.textContent = "Erreur : " + result.message;
                    statusMsg.className = "text-sm text-center mt-2 text-red-600 font-semibold";
                    submitBtn.disabled = false;
                    submitBtn.textContent = "Envoyer ma demande";
                }
            } catch (err) {
                statusMsg.textContent = "Erreur d'envoi : " + err.message;
                statusMsg.className = "text-sm text-center mt-2 text-red-600 font-semibold";
                submitBtn.disabled = false;
                submitBtn.textContent = "Envoyer ma demande";
            }
        }
